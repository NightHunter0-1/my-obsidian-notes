SELECT department_name
FROM departments
WHERE department_name REGEXP '^[^AEIOUY0-9aeiouy]';

SELECT 
    r.name AS region,
    COUNT(c.name) AS country_count
FROM regions r
LEFT JOIN countries c ON r.id = c.region_id
GROUP BY r.id, r.name
ORDER BY country_count DESC
LIMIT 4;
